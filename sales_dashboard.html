☁️ Google Sheets Sync Setup
Once set up, your entire sales dashboard syncs to Google Sheets. Your team loads it from any browser.
Step 1 — Create a Google Sheet, copy the Sheet ID from the URL
Step 2 — Extensions → Apps Script → paste this code:
const SHEET_ID = 'YOUR_SHEET_ID_HERE'; function doGet(e){ try{ const ss=SpreadsheetApp.openById(SHEET_ID); const sh=ss.getSheetByName('DashboardJSON'); if(sh&&sh.getLastRow()>0){ const chunks=sh.getRange(1,1,sh.getLastRow(),1).getValues(); const json=chunks.map(r=>r[0]).join(''); return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput( JSON.stringify({error:'No data yet. Click Save to Sheets first.'}) ); }catch(err){ return ContentService.createTextOutput(JSON.stringify({error:err.message})); } } function doPost(e){ try{ const p=JSON.parse(e.postData.contents); const ss=SpreadsheetApp.openById(SHEET_ID); if(p.chunks){ let sh=ss.getSheetByName('DashboardJSON'); if(!sh) sh=ss.insertSheet('DashboardJSON'); sh.clearContents(); sh.getRange(1,1,p.chunks.length,1).setValues(p.chunks.map(c=>[c])); } return ContentService.createTextOutput('ok'); }catch(err){ return ContentService.createTextOutput(JSON.stringify({error:err.message})); } }
Step 3 — Deploy → Web app → Execute as Me → Anyone → copy URL
Step 4 — Paste URL in sync panel → click ⬆ Save to Sheets
